# ============================================
# ShopSphere on AKS - Azure CLI commands
# Run in Azure Cloud Shell (Bash) - no az login needed
# ============================================

# 0. One-time setup: register resource providers (new subscriptions need this)
az provider register --namespace Microsoft.ContainerRegistry
az provider register --namespace Microsoft.ContainerService
az provider register --namespace Microsoft.Compute
az provider register --namespace Microsoft.Network
az provider register --namespace Microsoft.OperationalInsights

# Check status (wait until "Registered")
az provider show --namespace Microsoft.ContainerRegistry --query registrationState -o tsv

# 1. Create a Resource Group

az group create --name shopsphere-aks-rg --location italynorth

# 2. Create Azure Container Registry (name must be globally unique, lowercase, no dashes)
az acr create --resource-group shopsphere-aks-rg --name shopsphereacrfs --sku Basic

# Show registries and their login server
az acr list -o table

# 3. Enable admin user on ACR (needed for imagePullSecret)
az acr update -n shopsphereacrfs --admin-enabled true

# Show ACR credentials (NEVER commit these!)
az acr credential show -n shopsphereacrfs

# 4. Get the project code into Cloud Shell
cd ~
git clone https://github.com/FatemehSafatian/shopsphere-platform.git
cd ~/shopsphere-platform/ShopSphere

# 5. Build images in Azure and push to ACR (Cloud Shell has no Docker)
az acr build --registry shopsphereacrfs --image shopsphere-api:v1 --file ShopSphere.API/Dockerfile .
az acr build --registry shopsphereacrfs --image shopsphere-client:v1 --file ShopSphere.Client/Dockerfile .

# Alternative (if ACR Tasks is not allowed): import images built by GitHub Actions from Docker Hub
# az acr import --name shopsphereacrfs --source docker.io/fsafa/shopsphere-api:latest --image shopsphere-api:v1
# az acr import --name shopsphereacrfs --source docker.io/fsafa/shopsphere-app:latest --image shopsphere-client:v1

# Verify images in ACR
az acr repository list -n shopsphereacrfs -o table
az acr repository show-tags -n shopsphereacrfs --repository shopsphere-api -o table

# =====================================================================
# 6) Variables for the AKS steps (re-run these in every new Cloud Shell session)
# =====================================================================
RG=shopsphere-aks-rg
LOCATION=italynorth
ACR_NAME=shopsphereacrfs
AKS=shopsphere-aks
NODE_SIZE=Standard_D2s_v6

# =====================================================================
# 7) (Optional) Check which VM size can be used for the AKS node
#    A size is usable only if:
#      - it exists in the region (list-skus, Restrictions = None)
#      - its family has quota > 0 (list-usage)
#    Result for my subscription in italynorth:
#      D2s_v5 -> exists, but DSv5 family quota = 0  -> NOT usable
#      D2s_v6 -> exists, Dsv6 family quota = 10     -> usable
#      B2s    -> not offered in italynorth; B2s_v2 has quota 0
# =====================================================================
az vm list-skus --location italynorth --resource-type virtualMachines --size Standard_D2s_v --output table
az vm list-usage --location italynorth --output table | awk 'NR<=2 || $NF > 0'

# =====================================================================
# 8) Create the AKS cluster (1 node, free tier) 
# =====================================================================
az aks create \
  --resource-group shopsphere-aks-rg \
  --name shopsphere-aks \
  --location italynorth \
  --node-count 1 \
  --node-vm-size Standard_D2s_v6 \
  --node-osdisk-size 30 \
  --tier free \
  --generate-ssh-keys


# ============================================
# CLEANUP - run at the end of EVERY session!
# Deletes the resource group and everything inside it (ACR, AKS, ...)
# ============================================
az group delete --name shopsphere-aks-rg --yes --no-wait
